// -------------------------------------------------------------------------
// Le serveur JSON RdvMedecins doit être lancé (port 8080). On lui demande
// un captcha : GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - OkHttp, le client HTTP le plus utilisé sur Android : on décrit la
//     requête (Request), le client l'exécute (newCall(...).execute()) et

// ce que renvoie GET /api/captcha
data class Captcha(val id: String, val svg: String)

// un client HTTP : un objet coûteux à créer, qu'on crée UNE fois
val client = OkHttpClient()

// la requête : une fonction SUSPENDUE, qui renvoie le captcha ou lance une exception
suspend fun lireCaptcha(): Captcha = withContext(Dispatchers.IO) {
    val requete = Request.Builder().url("${Config.urlServeur}/api/captcha").build()
    client.newCall(requete).execute().use { reponse ->
        // OkHttp ne lance PAS d'exception pour une réponse d'erreur (4xx,
        // 5xx) : on teste le code de statut
        if (!reponse.isSuccessful) {
            throw Exception("le serveur a répondu ${reponse.code} ${reponse.message}")
        }
        val json = Json.parseToJsonElement(reponse.body!!.string()).jsonObject
        Captcha(json["id"]!!.jsonPrimitive.content, json["svg"]!!.jsonPrimitive.content)
    }
}

@Composable
fun PageCaptcha() {
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var enCours by remember { mutableStateOf(true) }
    var erreur by remember { mutableStateOf<String?>(null) }
    var nbRequetes by remember { mutableIntStateOf(0) }
    // le numéro de la DEMANDE : la changer relance l'effet ci-dessous
    var demande by remember { mutableIntStateOf(0) }

    // la requête, dès l'apparition de l'écran (demande = 0), puis à chaque
    // nouvelle demande. Si une demande arrive avant la réponse de la
    // précédente, la coroutine précédente est ANNULÉE : sa réponse, périmée,
    // ne sera jamais affichée
    LaunchedEffect(demande) {
        enCours = true
        erreur = null
        try {
            captcha = lireCaptcha()
            nbRequetes++
        } catch (e: CancellationException) {
            throw e // la coroutine a été annulée (nouvelle demande) : rien à afficher
        } catch (e: Exception) {
            // pas de réponse du tout (serveur arrêté, mauvaise adresse...), ou erreur
            erreur = e.message ?: e.toString()
            captcha = null
        }
        enCours = false
    }

    PageExemple("Exemple 22 - une première requête HTTP") {
        Text("Serveur : ${Config.urlServeur}", color = Color.Gray)
        Spacer(Modifier.height(12.dp))
        if (enCours) LinearProgressIndicator(Modifier.fillMaxWidth())
        erreur?.let {
            Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFFFEBEE))) {
                Text("Impossible d'obtenir un captcha : $it.\nLe serveur JSON RdvMedecins est-il lancé (port 8080) ?", Modifier.padding(12.dp))
            }
        }
        val c = captcha
        if (c != null && !enCours) {
            ImageSvg(
                c.svg,
                hauteur = 60.dp,
                modifier = Modifier.border(1.dp, Color(0xFFE0E0E0), RoundedCornerShape(6.dp)).padding(4.dp),
            )
            Spacer(Modifier.height(8.dp))
            Text("id : ${c.id}", fontFamily = FontFamily.Monospace, fontSize = 11.sp)
        }
        Spacer(Modifier.height(12.dp))
        Button(onClick = { demande++ }, enabled = !enCours) { Text("Autre image") }
        Spacer(Modifier.height(8.dp))
        Text("$nbRequetes requête(s) réussie(s)", color = Color.Gray)
    }
}

// commun/ImageSvg.kt - dessiner une image SVG donnée sous forme de texte
// -------------------------------------------------------------------------
// L'image du captcha est fabriquée par le serveur, au format SVG (un texte
// qui décrit un dessin). Android ne sait pas l'afficher seul : la
// bibliothèque AndroidSVG la dessine dans une image (un Bitmap), que
// Compose affiche avec le composant Image.

package fr.stahe.exemples.commun

import android.graphics.Bitmap
import android.graphics.Canvas
import androidx.compose.foundation.Image
import androidx.compose.foundation.layout.height
import androidx.compose.runtime.Composable
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.asImageBitmap
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.caverock.androidsvg.SVG

@Composable
fun ImageSvg(svg: String, modifier: Modifier = Modifier, hauteur: Dp = 50.dp, description: String? = null) {
    // le dessin n'est refait que si le texte SVG change (remember(svg))
    val image = remember(svg) {
        val document = SVG.getFromString(svg)
        // dessiné 3 fois plus grand que sa taille "naturelle" : une image nette
        val echelle = 3f
        val largeur = (document.documentWidth.takeIf { it > 0 } ?: 150f)
        val haut = (document.documentHeight.takeIf { it > 0 } ?: 50f)
        val bitmap = Bitmap.createBitmap((largeur * echelle).toInt(), (haut * echelle).toInt(), Bitmap.Config.ARGB_8888)
        val toile = Canvas(bitmap)
        toile.scale(echelle, echelle)
        document.renderToCanvas(toile)
        bitmap.asImageBitmap()
    }
    Image(image, contentDescription = description, contentScale = ContentScale.Fit, modifier = modifier.height(hauteur))
}

    // =========================================================================
// commun/ClientHttp.kt - un client HTTP qui gère les COOKIES
// -------------------------------------------------------------------------
// Le serveur RdvMedecins dépose le jeton JWT dans un COOKIE (httpOnly).
// Dans un navigateur, c'est le navigateur qui range les cookies et les
// renvoie avec chaque requête : le code de l'application n'a rien à faire.
// Sur un téléphone, il n'y a PAS de navigateur : c'est à l'application de
// jouer son rôle, c'est-à-dire :
//   1. lire l'entête "Set-Cookie" de chaque réponse, et ranger les cookies ;
//   2. les renvoyer dans l'entête "Cookie" de chaque requête.
// La bibliothèque OkHttp le fait pour nous, si on lui fournit un CookieJar
// (un "bocal à cookies") : une classe à deux fonctions, saveFromResponse
// (ranger) et loadForRequest (ressortir). Celui-ci range aussi les cookies
// sur l'appareil (SharedPreferences) : on reste connecté quand on arrête
// puis relance l'application.
// =========================================================================
package fr.stahe.exemples.commun

import android.content.Context
import android.content.SharedPreferences
import androidx.core.content.edit
import okhttp3.Cookie
import okhttp3.CookieJar
import okhttp3.HttpUrl
import okhttp3.HttpUrl.Companion.toHttpUrl
import okhttp3.OkHttpClient
import java.util.concurrent.TimeUnit

class CookiesSurLAppareil(private val stockage: SharedPreferences) : CookieJar {
    // nom du cookie -> le cookie (sa valeur, son domaine, son expiration...)
    private val cookies = mutableMapOf<String, Cookie>()

    init {
        // les cookies mémorisés lors d'une utilisation précédente : chacun
        // est rangé sous la forme de son entête Set-Cookie
        val base = Config.urlServeur.toHttpUrl()
        for (texte in stockage.getStringSet(CLE, emptySet()).orEmpty()) {
            Cookie.parse(base, texte)?.let { cookies[it.name] = it }
        }
    }

    // la réponse apporte des cookies : on les range. Un cookie déjà expiré
    // (déconnexion : le serveur l'efface ainsi) est retiré
    @Synchronized
    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
        for (c in cookies) {
            if (c.expiresAt < System.currentTimeMillis() || c.value.isEmpty()) {
                this.cookies.remove(c.name)
            } else {
                this.cookies[c.name] = c
            }
        }
        stockage.edit { putStringSet(CLE, this@CookiesSurLAppareil.cookies.values.map { it.toString() }.toSet()) }
    }

    // une requête part : les cookies qui la concernent (adresse, chemin), non expirés
    @Synchronized
    override fun loadForRequest(url: HttpUrl): List<Cookie> =
        cookies.values.filter { it.expiresAt >= System.currentTimeMillis() && it.matches(url) }

    companion object {
        private const val CLE = "rdvmedecins.cookies"
    }
}

// le client HTTP des exemples : un seul pour toute l'application
fun creerClientHttp(context: Context): OkHttpClient = OkHttpClient.Builder()
    .cookieJar(CookiesSurLAppareil(context.getSharedPreferences("exemples", Context.MODE_PRIVATE)))
    // pas de réponse au bout de 10 secondes : la requête échoue
    .connectTimeout(10, TimeUnit.SECONDS)
    .readTimeout(10, TimeUnit.SECONDS)
    .build()

// ex23/Commande.kt - une ACTION, son état, son résultat
// -------------------------------------------------------------------------
// Une action asynchrone (se connecter, enregistrer...) a un état que
// l'écran doit montrer : est-elle EN COURS (bouton désactivé) ? s'est-elle
// terminée par une ERREUR (message) ? C'est le modèle "Command", réduit à
// l'essentiel :
//
//     val connexion = remember { Commande() }
//     portee.launch { connexion.executer { seConnecter() } }   // lance l'action
//     connexion.enCours, connexion.erreur                      // son état
//
// Ses deux propriétés sont des états observés par Compose : l'écran se
// redessine quand l'action démarre et quand elle se termine.
// =========================================================================
package fr.stahe.exemples.ex23

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import kotlinx.coroutines.CancellationException

class Commande {
    var enCours by mutableStateOf(false)
        private set
    var erreur by mutableStateOf<String?>(null)
        private set

    // exécute l'action, si une autre n'est pas déjà en cours
    suspend fun executer(action: suspend () -> Unit) {
        if (enCours) return // pas de double envoi (double appui sur le bouton)
        enCours = true
        erreur = null
        try {
            action()
        } catch (e: CancellationException) {
            throw e // une coroutine annulée doit le rester
        } catch (e: Exception) {
            erreur = e.message ?: e.toString() // l'erreur est GARDÉE, pour être affichée
        } finally {
            // finally : exécuté dans TOUS les cas, succès ou erreur
            enCours = false

// commun/ErreurApi.kt - UNE classe pour toutes les erreurs de l'API
// -------------------------------------------------------------------------
// Toutes les réponses d'erreur du serveur RdvMedecins ont la même forme :
//   { "statusCode": 409, "cle": "ERRORS.LOGIN_TAKEN", "params": {...}, "champs": {...} }
// Mais une requête peut aussi échouer AUTREMENT : pas de réponse du tout
// (réseau coupé, serveur arrêté, mauvaise adresse), ou une réponse qui ne
// vient pas de notre serveur (une page d'erreur d'un proxy). On convertit
// tout cela en UNE classe d'exception, ErreurApi, qui porte la clé de
// traduction envoyée par le serveur (et, s'il y en a, ses paramètres et
// l'erreur de chaque champ).
// =========================================================================
package fr.stahe.exemples.commun

import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import kotlinx.serialization.json.jsonObject

class ErreurApi(
    val statusCode: Int, // 0 : le serveur n'a pas répondu
    val cle: String, // ex : "ERRORS.LOGIN_TAKEN"
    val params: Map<String, String> = emptyMap(),
    val champs: Map<String, String> = emptyMap(),
) : Exception("$statusCode $cle") {

    override fun toString() = "$statusCode $cle"

    // companion object : ce qui appartient à la CLASSE (les "static" de Java)
    companion object {
        // une réponse d'erreur (4xx, 5xx) -> une ErreurApi
        fun depuisReponse(statusCode: Int, corps: String): ErreurApi {
            // runCatching : le résultat, ou null si le corps n'est pas du JSON
            // (ce n'est alors pas notre serveur qui a répondu)
            val json = runCatching { Json.parseToJsonElement(corps).jsonObject }.getOrNull()
            val cle = (json?.get("cle") as? JsonPrimitive)?.content
            if (json == null || cle == null) {
                return ErreurApi(statusCode, if (statusCode >= 500) "ERRORS.SERVER_UNREACHABLE" else "ERRORS.SERVER_ERROR")
            }
            return ErreurApi(statusCode, cle, chaines(json["params"]), chaines(json["champs"]))
        }

        // un objet JSON { "nom": "ERRORS..." } -> Map<String, String> (vide sinon)
        private fun chaines(json: Any?): Map<String, String> =
            (json as? JsonObject)?.mapValues { (_, v) -> (v as? JsonPrimitive)?.content ?: v.toString() } ?: emptyMap()
    }
}

class Api(private val client: OkHttpClient) {

    suspend fun requete(methode: String, url: String, corps: JsonElement? = null): JsonElement? =
        withContext(Dispatchers.IO) {
            // un corps JSON (le serveur refuse un POST qui n'en est pas un)
            val corpsHttp = corps?.toString()?.toRequestBody("application/json".toMediaType())
            val requete = Request.Builder().url("${Config.urlServeur}/api$url").method(methode, corpsHttp).build()
            try {
                // use : la réponse est fermée (libérée) à la fin du bloc
                client.newCall(requete).execute().use { reponse ->
                    val texte = reponse.body?.string().orEmpty()
                    if (!reponse.isSuccessful) throw ErreurApi.depuisReponse(reponse.code, texte)
                    if (texte.isEmpty()) null else Json.parseToJsonElement(texte)
                }
            } catch (e: IOException) {
                // AUCUNE réponse (réseau, serveur arrêté, mauvaise adresse)
                throw ErreurApi(0, "ERRORS.SERVER_UNREACHABLE")
            }
        }

    suspend fun get(url: String) = requete("GET", url)

    // un POST a toujours un corps JSON : {} par défaut
    suspend fun post(url: String, corps: JsonElement = JsonObject(emptyMap())) = requete("POST", url, corps)

    suspend fun put(url: String, corps: JsonElement) = requete("PUT", url, corps)

    suspend fun delete(url: String, corps: JsonElement? = null) = requete("DELETE", url, corps)
}

                message = Message(false, e.toString())
            }
            chargement = false
        }
    }

@Composable
fun BarreConnexion(api: Api, onUtilisateurChange: (Utilisateur?) -> Unit) {
    var utilisateur by remember { mutableStateOf<Utilisateur?>(null) }
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var login by rememberSaveable { mutableStateOf("admin") }
    var password by rememberSaveable { mutableStateOf("") }
    var reponseCaptcha by rememberSaveable { mutableStateOf("") }
    var erreur by remember { mutableStateOf<String?>(null) }
    val portee = rememberCoroutineScope()

    fun changer(u: Utilisateur?) {
        utilisateur = u
        onUtilisateurChange(u)
    }

    suspend fun nouveauCaptcha() {
        captcha = api.get("/captcha").lire<Captcha>()
        reponseCaptcha = ""
    }

    // au démarrage : sommes-nous déjà connectés ?
    LaunchedEffect(Unit) {
        try {
            changer(api.get("/auth/moi").lire<Utilisateur>())
        } catch (e: ErreurApi) {
            changer(null) // 401 : personne n'est connecté
            try {
                nouveauCaptcha()
            } catch (e: ErreurApi) {
                erreur = "Le serveur JSON est-il lancé (${Config.urlServeur}) ?"
            }
        }
    }

    suspend fun seConnecter() {
        erreur = null
        try {
            val corps = buildJsonObject {
                put("login", login.trim())
                put("password", password)
                put("captchaId", captcha?.id)
                put("captcha", reponseCaptcha)
            }
            val u = api.post("/auth/login", corps).lire<Utilisateur>()
            password = ""
            changer(u)
        } catch (e: ErreurApi) {
            erreur = e.toString() // "401 LOGIN.ERROR"
            runCatching { nouveauCaptcha() }
        }
    }

    suspend fun seDeconnecter() {
        runCatching { api.post("/auth/logout") }
        changer(null)
        runCatching { nouveauCaptcha() }
    }

    Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFECEFF1))) {
        val u = utilisateur
        if (u != null) {
            Row(Modifier.padding(8.dp), verticalAlignment = Alignment.CenterVertically) {
                Icon(Icons.Default.Person, contentDescription = null)
                Spacer(Modifier.width(8.dp))
                Text("${u.nom} (${u.role})", Modifier.weight(1f))
                TextButton(onClick = { portee.launch { seDeconnecter() } }) {
                    Icon(Icons.AutoMirrored.Filled.ExitToApp, contentDescription = null)
                    Text("Se déconnecter")
                }
            }
        } else {
            Column(Modifier.padding(8.dp)) {
                Row {
                    OutlinedTextField(login, { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.weight(1f))
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(
                        password,
                        { password = it },
                        label = { Text("mot de passe") },
                        singleLine = true,
                        visualTransformation = PasswordVisualTransformation(),
                        modifier = Modifier.weight(1f),
                    )
                }
                Row(verticalAlignment = Alignment.CenterVertically) {
                    // l'image SVG du captcha, fabriquée par NOTRE serveur (cf. exemple 22)
                    captcha?.let { ImageSvg(it.svg, hauteur = 40.dp) }
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(reponseCaptcha, { reponseCaptcha = it }, label = { Text("captcha") }, singleLine = true, modifier = Modifier.weight(1f))
                    IconButton(onClick = { portee.launch { seConnecter() } }) {
                        Icon(Icons.Default.Check, contentDescription = "Se connecter")
                    }
                }
                erreur?.let { Text(it, color = Color.Red) }
            }
        }
    }
}

class Dictionnaire(context: Context, fichier: String) {
    // le contenu du fichier JSON, lu une fois pour toutes
    private val racine: JsonObject = context.assets.open(fichier).bufferedReader().use { lecteur ->
        Json.parseToJsonElement(lecteur.readText()) as JsonObject
    }

    // "FORM.AJOUTE" -> racine["FORM"]["AJOUTE"] ; null si absente
    private fun chercher(cle: String): String? {
        var noeud: JsonElement? = racine
        for (partie in cle.split('.')) {
            noeud = (noeud as? JsonObject)?.get(partie)
        }
        return (noeud as? JsonPrimitive)?.takeIf { it.isString }?.content
    }

    fun t(cle: String, params: Map<String, String> = emptyMap()): String {
        val texte = chercher(cle) ?: return cle
        // {{nom}} -> la valeur du paramètre nom (ou {{nom}} s'il n'est pas fourni)
        return Regex("\\{\\{(\\w+)\\}\\}").replace(texte) { m -> params[m.groupValues[1]] ?: m.value }
    }
}

    val context = LocalContext.current
    val langue = LocalConfiguration.current.locales[0].language.let { if (it == "en") "en" else "fr" }
    // relu seulement si la langue change
    return remember(langue) { Dictionnaire(context, "i18n/${prefixe}_$langue.json") }
}

class ClientsModele(private val api: Api) : ViewModel() {

        private set
    var clients by mutableStateOf(listOf<Personne>())
        private set

    // le formulaire : idEdite == null -> création, sinon modification
    var idEdite by mutableStateOf<Int?>(null)
        private set
    private var version = 0

    // les valeurs saisies, par nom de champ (les noms attendus par le serveur)
    var saisie by mutableStateOf(vide)
        private set

    // l'erreur de chaque champ (une CLÉ de traduction), renvoyée par le serveur
    var erreurs by mutableStateOf(mapOf<String, String>())
        private set

    // le message général de la dernière erreur, et celui d'un succès
    var erreurPage by mutableStateOf<ErreurApi?>(null)
        private set
    var succes by mutableStateOf<MessageCle?>(null)
        private set

    val envoi = Commande()

        utilisateur = u
        clients = emptyList()
        if (u?.role == Role.ADMIN) recharger()
    }

    private suspend fun recharger() {
        clients = try {
            api.get("/clients").lire()
        } catch (e: ErreurApi) {
            emptyList()
        }
    }

    private fun effacerMessages() {
        erreurs = emptyMap()
        erreurPage = null
        succes = null
    }

    // un champ modifié : sa valeur change, son erreur disparaît
    fun modifier(champ: String, valeur: String) {
        saisie = saisie + (champ to valeur)
        erreurs = erreurs - champ
    }

    fun nouveau() {
        idEdite = null
        version = 0
        saisie = vide
        effacerMessages()
    }

    fun editer(c: Personne) {
        idEdite = c.id
        // la VERSION lue avec la fiche repart avec la modification : si
        // quelqu'un a modifié la fiche entre-temps, le serveur répondra 409
        // CONCURRENT_UPDATE
        version = c.version
        saisie = mapOf("titre" to c.titre, "prenom" to c.prenom, "nom" to c.nom, "login" to (c.login ?: ""), "password" to "")
        effacerMessages()
    }

    suspend fun enregistrer() {
        effacerMessages()
        val corps = buildJsonObject {
            for ((champ, valeur) in saisie) put(champ, valeur)
            put("version", version)
        }
        try {
            val id = idEdite
            val client: Personne = (if (id == null) api.post("/clients", corps) else api.put("/clients/$id", corps)).lire()
            val cle = if (id == null) "FORM.AJOUTE" else "FORM.MODIFIE"
            nouveau() // formulaire vidé, messages effacés...
            // ... sauf celui-ci
            succes = MessageCle(cle, mapOf("nom" to "${client.prenom} ${client.nom}", "login" to (client.login ?: "")))
            recharger()
        } catch (e: ErreurApi) {
            erreurPage = e // le message général
            erreurs = e.champs // l'erreur de chaque champ
        }
    }

    suspend fun supprimer(c: Personne) {
        effacerMessages()
        try {
            api.delete("/clients/${c.id}")
            succes = MessageCle("FORM.SUPPRIME", mapOf("nom" to "${c.prenom} ${c.nom}"))
        } catch (e: ErreurApi) {
            erreurPage = e
        }
        recharger()
    }

    companion object {
        val vide = mapOf("titre" to "Mr", "prenom" to "", "nom" to "", "login" to "", "password" to "")
    }
}
