    5. La navigation

    Une application a plusieurs écrans. Sur [Android], avec [Compose], on n'ouvre pas une activité par écran : une SEULE activité affiche, tour à tour, les écrans de l'application, sous le contrôle d'un ROUTEUR. C'est la bibliothèque [navigation-compose] de [Jetpack], l'équivalent du paquet [go_router] de [Flutter].

    // =========================================================================
// ex17/Routes.kt - les ROUTES de l'application
// -------------------------------------------------------------------------
// Une application a plusieurs écrans (des "pages"). Le ROUTEUR associe une
// ROUTE à chaque écran. La bibliothèque Navigation de Jetpack (navigation-
// compose) gère une PILE d'écrans : on empile un écran (navigate), le
// bouton Retour du téléphone le dépile (popBackStack).
// Une route est un objet Kotlin, @Serializable :
//   - un "object" pour un écran sans paramètre (Accueil) ;
//   - une "data class" pour un écran avec paramètres : DetailMedecin(id = 2).
// Le type des paramètres est vérifié à la compilation : DetailMedecin("2")
// serait refusé. (Navigation sait ranger ces objets dans la pile, d'où
// @Serializable.)
// Chaque écran peut aussi avoir une ADRESSE, un "lien profond" (deep link)
// du type app://exemples/medecins/2 : c'est ce qui se rapproche le plus,
// sur Android, des adresses du web (cf. Ex17Activity.kt).
// =========================================================================
package fr.stahe.exemples.ex17

import kotlinx.serialization.Serializable

@Serializable
object Accueil

@Serializable
object ListeMedecins

// un paramètre de route
@Serializable
data class DetailMedecin(val id: Int)

// un paramètre FACULTATIF (avec une valeur par défaut) : ?q=... dans l'adresse
@Serializable
data class Recherche(val q: String = "")

@Serializable
object APropos

// une ancienne adresse, qui REDIRIGE vers la liste des médecins
@Serializable
object Docteurs

// l'écran des adresses inconnues
@Serializable
data class PageIntrouvable(val adresse: String)

// le début de toutes les adresses de cet exemple
const val BASE = "app://exemples"

Le type des paramètres est vérifié à la compilation ([DetailMedecin("2")] serait refusé) ; [@Serializable] : [Navigation] sait ranger ces objets dans la pile, et les reconstituer si l'activité est recréée ;
    • ligne 34 : un paramètre facultatif, avec une valeur par défaut ;
    • ligne 48 : chaque écran peut avoir une ADRESSE, un « lien profond » (deep link) de la forme [app://exemples/medecins/2] : c'est ce qui se rapproche le plus, sur [Android], des adresses du web.

    // =========================================================================
// ex17/Ex17Activity.kt - le NavHost : les routes et leurs écrans
// -------------------------------------------------------------------------
//   - rememberNavController() : le contrôleur de navigation (la pile) ;
//   - NavHost(nav, startDestination = Accueil) { ... } : l'écran de la
//     route courante. Le bloc décrit le GRAPHE de navigation : une
//     entrée composable<Route> { ... } par route ;
//   - entree.toRoute<DetailMedecin>() : l'objet route de l'écran, avec
//     ses paramètres ;
//   - deepLinks : l'adresse de l'écran. Grâce au filtre d'intentions de
//     l'activité (AndroidManifest.xml), une adresse app://exemples/...
//     peut même être ouverte de l'extérieur de l'application :
//         adb shell am start -a android.intent.action.VIEW -d "app://exemples/medecins/2"
// =========================================================================

import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.activity.enableEdgeToEdge

import androidx.compose.runtime.LaunchedEffect
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable

import androidx.navigation.navDeepLink
import androidx.navigation.toRoute
import fr.stahe.exemples.commun.ThemeExemples
import fr.stahe.exemples.ex17.pages.PageAPropos

class Ex17Activity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { ThemeExemples { Application() } }
    }
}

@Composable
fun Application() {
    val nav = rememberNavController()
    Coquille(nav) {
        NavHost(navController = nav, startDestination = Accueil) {
            composable<Accueil>(deepLinks = listOf(navDeepLink<Accueil>(basePath = "$BASE/accueil"))) {
                PageAccueil(onLien = { nav.ouvrirLien(it) }, onRedirection = { nav.navigate(Docteurs) })
            }
            composable<ListeMedecins>(deepLinks = listOf(navDeepLink<ListeMedecins>(basePath = "$BASE/medecins"))) {
                // navigate : l'écran du médecin est EMPILÉ au-dessus de la liste
                PageListeMedecins(onVoir = { id -> nav.navigate(DetailMedecin(id)) })
            }
            // app://exemples/medecins/2 : le paramètre id est pris dans l'adresse
            composable<DetailMedecin>(deepLinks = listOf(navDeepLink<DetailMedecin>(basePath = "$BASE/medecins"))) { entree ->
                val route = entree.toRoute<DetailMedecin>()
                PageDetailMedecin(
                    id = route.id,
                    // popBackStack : dépiler l'écran (comme le bouton Retour) ; s'il
                    // n'y a rien dessous (écran ouvert par une adresse), aller à la liste
                    onRetour = { if (!nav.popBackStack()) nav.navigate(ListeMedecins) },
                    // REMPLACER l'écran courant (pas d'empilement) : on dépile le
                    // médecin courant (inclusive) avant d'empiler le suivant
                    onSuivant = { suivant ->
                        nav.navigate(DetailMedecin(suivant)) { popUpTo<DetailMedecin> { inclusive = true } }
                    },
                )
            }
            // app://exemples/recherche?q=pédiatre
            composable<Recherche>(deepLinks = listOf(navDeepLink<Recherche>(basePath = "$BASE/recherche"))) { entree ->
                PageRecherche(q = entree.toRoute<Recherche>().q, onRechercher = { saisie -> nav.navigate(Recherche(saisie)) })
            }
            composable<APropos> { PageAPropos() }
            // une REDIRECTION : l'écran n'affiche rien, il remplace aussitôt
            // sa propre route par une autre
            composable<Docteurs> {
                LaunchedEffect(Unit) {
                    nav.navigate(ListeMedecins) { popUpTo<Docteurs> { inclusive = true } }
                }
            }
            composable<PageIntrouvable> { entree ->
                EcranIntrouvable(adresse = entree.toRoute<PageIntrouvable>().adresse, onAccueil = { nav.allerA(Accueil) })
            }
        }
    }
}
    • ligne 34 : [import ... as EcranIntrouvable] : un alias, car la fonction [PageIntrouvable] des écrans a le même nom que la route [PageIntrouvable] ;
    • lignes 46-48 : [rememberNavController()] : le contrôleur de navigation (la pile) ; [NavHost(nav, startDestination = Accueil)] affiche l'écran de la route courante ;
    • lignes 49-54 : une entrée [composable<Route>] par route, avec son adresse ([deepLinks]). Les écrans ne connaissent pas le routeur : ils reçoivent des FONCTIONS pour naviguer ([onLien], [onVoir]...) ;
    • lignes 57-58 : [entree.toRoute<DetailMedecin>()] : l'objet route, avec ses paramètres. Avec [navDeepLink<DetailMedecin>(basePath = ...)], le paramètre [id] est pris dans l'adresse ([.../medecins/2]) ;
    • ligne 63 : [popBackStack()] dépile l'écran, comme le bouton Retour, et renvoie [false] s'il n'y a rien dessous (écran ouvert par une adresse) ;
    • lignes 66-67 : REMPLACER l'écran courant : on dépile le médecin courant ([popUpTo<DetailMedecin> [ inclusive = true ]]) avant d'empiler le suivant ;
    • lignes 78-80 : une REDIRECTION : l'écran n'affiche rien, il remplace aussitôt sa route par une autre. La navigation est un « effet » ([LaunchedEffect]) : on ne navigue jamais PENDANT l'affichage.

// un onglet : la route où il mène, son libellé, son icône, et le test qui
// dit s'il est actif pour un écran donné
class Onglet(val route: Any, val libelle: String, val icone: ImageVector, val actif: (NavDestination) -> Boolean)

val onglets = listOf(
    Onglet(Accueil, "Accueil", Icons.Default.Home) { it.hasRoute<Accueil>() },
    // l'onglet Médecins reste actif sur l'écran d'un médecin
    Onglet(ListeMedecins, "Médecins", Icons.Default.Person) { it.hasRoute<ListeMedecins>() || it.hasRoute<DetailMedecin>() },
    Onglet(Recherche("généraliste"), "Recherche", Icons.Default.Search) { it.hasRoute<Recherche>() },
    Onglet(APropos, "À propos", Icons.Default.Info) { it.hasRoute<APropos>() },
)

// l'adresse de l'écran courant, pour l'afficher
fun adresseDe(entree: NavBackStackEntry?): String {
    val d = entree?.destination ?: return ""
    return when {
        d.hasRoute<Accueil>() -> "/"
        d.hasRoute<ListeMedecins>() -> "/medecins"
        d.hasRoute<DetailMedecin>() -> "/medecins/${entree.toRoute<DetailMedecin>().id}"
        d.hasRoute<Recherche>() -> "/recherche?q=${entree.toRoute<Recherche>().q}"
        d.hasRoute<APropos>() -> "/a-propos"
        d.hasRoute<PageIntrouvable>() -> entree.toRoute<PageIntrouvable>().adresse
        else -> "?"
    }
}

// aller à un onglet : la pile est REMPLACÉE par celle de cet onglet
fun NavHostController.allerA(route: Any) {
    navigate(route) {
        // on dépile tout, jusqu'au premier écran (exclu)...
        popUpTo(graph.findStartDestination().id)
        // ... et l'on n'empile pas deux fois le même écran
        launchSingleTop = true
    }
}

// suivre un LIEN (une adresse) ; une adresse inconnue mène à PageIntrouvable
fun NavHostController.ouvrirLien(adresse: String) {
    try {
        navigate(Uri.parse("$BASE$adresse"))
    } catch (e: IllegalArgumentException) {
        navigate(PageIntrouvable(adresse))
    }
}

@Composable
fun Coquille(nav: NavHostController, contenu: @Composable () -> Unit) {
    val entree by nav.currentBackStackEntryAsState()
    val destination = entree?.destination
    Scaffold(
        topBar = { TopAppBar(title = { Text("Exemple 17 - le routeur") }) },
        // NavigationBar : la barre d'onglets du bas, typique d'un mobile
        bottomBar = {
            NavigationBar {
                for (o in onglets) {
                    NavigationBarItem(
                        selected = destination != null && o.actif(destination),
                        onClick = { nav.allerA(o.route) },
                        icon = { Icon(o.icone, contentDescription = null) },
                        label = { Text(o.libelle) },
                    )
                }
            }
        },
    ) { marges ->
        Column(Modifier.padding(marges).fillMaxSize()) {
            Text(
                "adresse courante : ${adresseDe(entree)}",
                fontFamily = FontFamily.Monospace,
                fontSize = 12.sp,
                modifier = Modifier.fillMaxWidth().background(Color(0xFFF5F5F5)).padding(8.dp),
            )
            Box(Modifier.padding(16.dp).weight(1f)) { contenu() }
        }
    }
}

@Composable
fun PageListeMedecins(onVoir: (Int) -> Unit) {

    }

// une page à paramètre : le routeur lui passe id, un Int
@Composable
fun PageDetailMedecin(id: Int, onRetour: () -> Unit, onSuivant: (Int) -> Unit) {
    // find : le premier médecin qui convient, ou null
    val medecin = medecins.find { it.id == id }
    if (medecin == null) {
        Text("Le médecin n° $id n'existe pas.", color = Color.Red)
        return
    }
    val suivant = medecin.id % medecins.size + 1

// -------------------------------------------------------------------------
// L'utilisateur connecté doit être connu des ÉCRANS (la barre du haut,
// l'agenda) et des GARDES du routeur. On le range dans un objet UNIQUE
// (object : une classe qui n'a qu'une instance, créée au premier usage) :
//   - une propriété d'état, observée par Compose (mutableStateOf) : tout

package fr.stahe.exemples.ex18

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue

enum class Role { ADMIN, DOCTOR, USER }

data class Utilisateur(val login: String, val role: Role)

// trois comptes "en dur" (mot de passe = login)
private val comptes = mapOf("admin" to Role.ADMIN, "pelissier" to Role.DOCTOR, "martin" to Role.USER)

object Session {
    // l'état : null, personne n'est connecté. private set : modifiable seulement ici
    var utilisateur by mutableStateOf<Utilisateur?>(null)
        private set

    fun connecter(login: String, password: String): Boolean {
        val role = comptes[login]
        if (role == null || password != login) {
            return false
        }
        utilisateur = Utilisateur(login, role)
        return true
    }

    fun deconnecter() {
        utilisateur = null
    }
}

// ex18/Routes.kt - des routes PROTÉGÉES par des gardes
// -------------------------------------------------------------------------
// Un GARDE est une fonction qui reçoit l'utilisateur connecté (ou null) et
// renvoie :
//   - null : l'écran peut s'afficher ;
//   - une route : l'écran est REMPLACÉ par cette route (une redirection).
// Le composant Protege(...) applique les gardes d'un écran. Comme il LIT
// l'utilisateur de la session (un état observé), il est réévalué à chaque
// connexion ou déconnexion : un écran qui n'est plus permis est quitté.
//
// C'est l'équivalent, côté client, des gardes NestJS (JwtAuthGuard,
// RolesGuard) - avec une différence FONDAMENTALE : un garde du client n'est
// qu'un confort. L'application est entre les mains de l'utilisateur, qui
// peut la modifier à volonté. La vraie protection des données reste celle
// du SERVEUR (cf. chapitre sur la boîte noire).
// =========================================================================
package fr.stahe.exemples.ex18

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.navigation.NavHostController
import kotlinx.serialization.Serializable

// refus : déposé par le garde roleRequis quand le rôle ne convient pas
@Serializable
data class Accueil(val refus: String? = null)

// redirect : l'adresse de l'écran qu'on voulait ouvrir
@Serializable
data class Connexion(val redirect: String? = null)

@Serializable
object Agenda

@Serializable
object Administration

const val BASE = "app://exemples18"

typealias Garde = (Utilisateur?) -> Any?

// le garde "connexion requise" : pas connecté -> à la page de connexion, en
// retenant l'adresse de l'écran demandé
fun connexionRequise(adresse: String): Garde = { u -> if (u == null) Connexion(redirect = adresse) else null }

// le garde "rôle requis" : connecté, mais pas avec le bon rôle -> retour à
// l'accueil, avec un motif
fun roleRequis(adresse: String, vararg roles: Role): Garde = { u ->
    if (u == null || u.role !in roles) Accueil(refus = adresse) else null
}

// les gardes, dans l'ordre : le premier qui renvoie une route l'emporte
@Composable
fun Protege(nav: NavHostController, vararg gardes: Garde, contenu: @Composable () -> Unit) {
    val redirection = gardes.firstNotNullOfOrNull { garde -> garde(Session.utilisateur) }
    if (redirection == null) {
        contenu()
    } else {
        // la navigation est un "effet" : on ne navigue pas PENDANT l'affichage
        LaunchedEffect(redirection) {
            val ici = nav.currentBackStackEntry?.destination?.id ?: return@LaunchedEffect
            // l'écran refusé est remplacé (inclusive) par la redirection
            nav.navigate(redirection) { popUpTo(ici) { inclusive = true } }
        }
    }
}

@Composable
fun Application() {
    val nav = rememberNavController()
    Coquille(nav) {
        NavHost(navController = nav, startDestination = Accueil()) {
            composable<Accueil> { entree -> PageAccueil(refus = entree.toRoute<Accueil>().refus) }
            composable<Connexion> { entree ->
                PageConnexion(
                    redirection = entree.toRoute<Connexion>().redirect,
                    onConnecte = { r ->
                        // on n'accepte qu'une adresse de CETTE application
                        // la pile est vidée (popUpTo le graphe entier) : la connexion
                        // n'y reste pas, Retour ne la rouvrira pas
                        val viderLaPile = navOptions { popUpTo(nav.graph.id) { inclusive = true } }
                        if (r != null && r.startsWith("$BASE/")) {
                            nav.navigate(Uri.parse(r), viderLaPile)
                        } else {
                            nav.navigate(Accueil(), viderLaPile)
                        }
                    },
                )
            }
            composable<Agenda>(deepLinks = listOf(navDeepLink<Agenda>(basePath = "$BASE/agenda"))) {
                Protege(nav, connexionRequise("$BASE/agenda")) { PageAgenda() }
            }
            composable<Administration>(deepLinks = listOf(navDeepLink<Administration>(basePath = "$BASE/administration"))) {
                Protege(nav, connexionRequise("$BASE/administration"), roleRequis("/administration", Role.ADMIN)) {
                    PageAdministration()
                }
            }
        }
    }
}

// la connexion. Après une connexion réussie, onConnecte reçoit l'adresse de
// l'écran qu'on voulait ouvrir (paramètre redirect déposé par le garde),
// ou null
// =========================================================================
@Composable
fun PageConnexion(redirection: String?, onConnecte: (String?) -> Unit) {
    var login by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var erreur by remember { mutableStateOf(false) }

    fun valider() {
        if (!Session.connecter(login.trim(), password)) {
            erreur = true
            return
        }
        onConnecte(redirection)
    }

    Column {
        if (redirection != null) {
            Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFFFF8E1))) {
                Text("Connectez-vous pour ouvrir ${redirection.removePrefix(BASE)}.", Modifier.padding(12.dp))
            }
        }
        OutlinedTextField(value = login, onValueChange = { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.fillMaxWidth())
        OutlinedTextField(
            value = password,
            onValueChange = { password = it },
            label = { Text("mot de passe") },
            singleLine = true,
            visualTransformation = PasswordVisualTransformation(),
            keyboardOptions = KeyboardOptions(imeAction = ImeAction.Go),
            keyboardActions = KeyboardActions(onGo = { valider() }),
            modifier = Modifier.fillMaxWidth(),
        )
        Spacer(Modifier.height(16.dp))
        Button(onClick = { valider() }) { Text("Se connecter") }
        if (erreur) {
            Text("Login ou mot de passe incorrect.", color = Color.Red, modifier = Modifier.padding(top = 8.dp))
        }
    }
}
