    4. Les composants

    Le chapitre précédent a présenté les composants de [Material 3] et l'état. Ce chapitre montre comment écrire SES PROPRES composants : leurs paramètres, la façon dont ils préviennent leur parent, la composition, leur cycle de vie, les données qu'on leur fournit de loin, les comportements qu'on réutilise. Avec [Compose], un composant est une simple fonction [@Composable] : l'équivalent des widgets de [Flutter].

// Il reçoit ses données de son parent par ses PARAMÈTRES. Leur type est
// vérifié à la compilation : CarteMedecin(medecin = 42, ...) serait refusé.
//   - un paramètre sans valeur par défaut : le parent DOIT le fournir ;
//   - compact: Boolean = false : un paramètre facultatif, avec sa valeur
//     par défaut ;
//   - modifier: Modifier = Modifier : la convention de Compose, le parent
//     peut ajuster la taille, les marges... de l'enfant.
//
// Règle d'or : un composant ne MODIFIE JAMAIS ses paramètres (ce sont des
// val). Les données descendent du parent vers l'enfant ; pour demander un
// changement, l'enfant appelle une fonction que le parent lui a passée
// (cf. exemple 11). Quand le parent change ses données, Compose RÉEXÉCUTE
// l'enfant avec de nouveaux paramètres.

// composant ne peut les voir, ni être gêné par elles
private val couleurInitiales = Color(0xFF087EA4)

@Composable
fun CarteMedecin(medecin: Medecin, rang: Int, modifier: Modifier = Modifier, compact: Boolean = false) {
    // une valeur calculée à partir des paramètres
    val initiales = "${medecin.prenom.first()}${medecin.nom.first()}"
    Card(
        modifier = modifier.padding(vertical = 4.dp),
        shape = RoundedCornerShape(12.dp),
        // une bordure verte pour un médecin disponible (null : pas de bordure)
        border = if (medecin.disponible) BorderStroke(2.dp, Color(0xFF4CAF50)) else null,
    ) {
        Row(Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
            // un cercle coloré, avec les initiales au centre
            Box(
                contentAlignment = Alignment.Center,
                modifier = Modifier.size(40.dp).background(couleurInitiales, CircleShape),
            ) {
                Text(initiales, color = Color.White, fontWeight = FontWeight.Bold)
            }
            Spacer(Modifier.width(12.dp))
            // weight(1f) : le texte prend toute la largeur restante
            Column(Modifier.weight(1f)) {
                Text("$rang. ${medecin.titre} ${medecin.prenom} ${medecin.nom}", style = MaterialTheme.typography.titleMedium)
                if (!compact) {
                    Text(medecin.specialite, color = Color.Gray)
                }
            }
        }
    }
}

@Composable
fun PageMedecins() {
    var medecins by remember {
        mutableStateOf(
            listOf(
                Medecin(1, "Mme", "Marie", "PELISSIER", "généraliste", true),
                Medecin(2, "Mr", "Paul", "TATOU", "pédiatre", false),
                Medecin(3, "Melle", "Sophie", "DUPONT", "généraliste", true),
            ),
        )
    }
    var compact by remember { mutableStateOf(false) }

    PageExemple("Exemple 10 - les composants et leurs paramètres") {
        ListItem(
            headlineContent = { Text("affichage compact") },
            trailingContent = { Switch(checked = compact, onCheckedChange = { compact = it }) },
        )
        // le parent change SES données : Compose réexécute cette fonction, qui
        // rappelle les enfants avec leurs nouveaux paramètres
        OutlinedButton(onClick = { medecins = medecins.map { it.copy(disponible = !it.disponible) } }) {
            Text("Inverser les disponibilités")
        }
        Spacer(Modifier.height(8.dp))
        medecins.forEachIndexed { i, m ->
            // chaque carte reçoit SES paramètres ; key : son identité (cf. exemple 04)
            key(m.id) {
                CarteMedecin(medecin = m, rang = i + 1, compact = compact)
            }
        }
        Spacer(Modifier.height(16.dp))
        Text("La même carte, en version compacte, en fixant le paramètre :", color = Color.Gray)
        // le parent fixe aussi la LARGEUR de l'enfant, par le paramètre modifier
        CarteMedecin(medecin = medecins[0], rang = 1, compact = true, modifier = Modifier.width(260.dp))
    }
}

// ex11/Compteur.kt - un enfant qui PRÉVIENT son parent
// -------------------------------------------------------------------------
// Le parent lui passe des FONCTIONS en paramètres (par convention, leur
// nom commence par "on"), et l'enfant les appelle quand il se passe
// quelque chose :
//     Compteur(libelle = "Cafés", max = 3, onIncremente = { ... }, onLimite = { ... })
// Le type d'une fonction s'écrit (paramètres) -> résultat :
//   - (Int) -> Unit : une fonction qui reçoit un Int et ne renvoie rien ;
//   - () -> Unit    : une fonction sans paramètre, qui ne renvoie rien.
// =========================================================================
package fr.stahe.exemples.ex11

import androidx.compose.foundation.layout.Row
import androidx.compose.foundation.layout.Spacer
import androidx.compose.foundation.layout.width
import androidx.compose.material3.OutlinedButton
import androidx.compose.material3.SuggestionChip
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableIntStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp

@Composable
fun Compteur(libelle: String, max: Int, onIncremente: (Int) -> Unit, onLimite: () -> Unit) {
    // l'état PROPRE à ce compteur : chaque Compteur a le sien
    var valeur by remember { mutableIntStateOf(0) }

    fun plus() {
        if (valeur >= max) {
            onLimite()
            return
        }
        valeur++
        // on PRÉVIENT le parent, en lui transmettant la nouvelle valeur
        onIncremente(valeur)
    }

    Row(verticalAlignment = Alignment.CenterVertically) {
        Text(libelle, Modifier.width(110.dp))
        OutlinedButton(onClick = { plus() }) { Text("+1") }
        Spacer(Modifier.width(12.dp))
        SuggestionChip(onClick = {}, label = { Text("$valeur / $max") })
    }
}
    • ligne 30 : le type d'une fonction s'écrit [(paramètres) -> résultat] : [(Int) -> Unit] reçoit un [Int] et ne renvoie rien ; [() -> Unit] n'a pas de paramètre ;
    • ligne 32 : l'état PROPRE à ce compteur : chaque [Compteur] a le sien ;
    • lignes 34-41 : arrivé au maximum, le compteur prévient le parent ([onLimite()]) ; sinon, il s'incrémente et transmet sa nouvelle valeur ([onIncremente(valeur)]).

// -------------------------------------------------------------------------
// Un champ de formulaire "maison" (libellé + champ + aide + compteur de
// caractères) qui s'utilise comme les champs de Compose :

package fr.stahe.exemples.ex11

import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.material3.OutlinedTextField
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier

@Composable
fun ChampTexte(valeur: String, onChange: (String) -> Unit, libelle: String, max: Int, aide: String = "") {
    val plein = valeur.length == max
    OutlinedTextField(
        value = valeur,
        // le champ refuse les caractères de trop : la frappe n'est transmise
        // au parent que si le texte ne dépasse pas max caractères
        onValueChange = { texte -> if (texte.length <= max) onChange(texte) },
        label = { Text(libelle) },
        singleLine = true,
        // la bordure devient rouge quand le champ est plein
        isError = plein,
        supportingText = { Text(if (plein) "Longueur maximale atteinte - ${valeur.length}/$max" else "$aide - ${valeur.length}/$max") },
        modifier = Modifier.fillMaxWidth(),
    )
}
    • ligne 28 : la frappe n'est transmise au parent que si le texte ne dépasse pas [max] caractères : le champ « refuse » les caractères de trop ;
    • lignes 32-33 : la bordure devient rouge quand le champ est plein, et le texte d'aide le dit.

@Composable
fun PageParent() {
    var total by remember { mutableIntStateOf(0) }
    var message by remember { mutableStateOf("") }
    var prenom by remember { mutableStateOf("Marie") }
    var nom by remember { mutableStateOf("PELISSIER") }

    // la fonction appelée par un compteur : elle reçoit la valeur qu'il transmet
    fun surIncremente(libelle: String, valeur: Int) {
        total++
        message = "$libelle vaut maintenant $valeur"
    }

    val styleTitre = MaterialTheme.typography.titleSmall
    PageExemple("Exemple 11 - l'enfant prévient son parent") {
        Card {
            Column(Modifier.padding(12.dp)) {
                Text("1. Des fonctions appelées par les enfants", style = styleTitre)
                Compteur(
                    libelle = "Cafés",
                    max = 3,
                    onIncremente = { v -> surIncremente("Cafés", v) },
                    onLimite = { message = "Pas plus de 3 cafés !" },
                )
                Compteur(
                    libelle = "Verres d'eau",
                    max = 8,
                    onIncremente = { v -> surIncremente("Verres d'eau", v) },
                    onLimite = { message = "Assez bu." },
                )
                Spacer(Modifier.height(8.dp))
                Text("Total des clics : $total - $message", fontStyle = FontStyle.Italic)
            }
        }
        Card {
            Column(Modifier.padding(12.dp)) {
                Text("2. Des composants contrôlés", style = styleTitre)
                ChampTexte(valeur = prenom, onChange = { prenom = it }, libelle = "Prénom", max = 20, aide = "Obligatoire")
                ChampTexte(valeur = nom, onChange = { nom = it }, libelle = "Nom", max = 20, aide = "Mis en majuscules à l'affichage")
                Spacer(Modifier.height(8.dp))
                Text(
                    buildAnnotatedString {
                        append("Données du parent : ")
                        withStyle(SpanStyle(fontWeight = FontWeight.Bold)) { append("$prenom ${nom.uppercase()}") }
                    },
                )
                Spacer(Modifier.height(8.dp))
                OutlinedButton(onClick = {
                    prenom = ""
                    nom = ""
                }) { Text("Vider (depuis le parent)") }
            }
        }
    }
}

// -------------------------------------------------------------------------
// Les paramètres d'un composant transmettent des DONNÉES... mais aussi du
// CONTENU : un paramètre peut être une fonction @Composable, que le

// -------------------------------------------------------------------------
// La vie d'un composant dans l'écran ("la composition") :
//   1. il ENTRE dans la composition : sa fonction est exécutée pour la
//      première fois. On y démarre ce qui doit l'être ;
//   2. il est RECOMPOSÉ autant de fois qu'il le faut (son état ou ses

//   3. il SORT de la composition : il disparaît de l'écran, pour toujours.
//      On y ARRÊTE ce qui a été démarré.

// l'affichage proprement dit :
//   - LaunchedEffect(cle) { ... } : lance une COROUTINE (du code qui peut
//     attendre, cf. exemple 19) à l'entrée dans la composition, la relance
//     si cle change, l'ANNULE à la sortie ;
//   - DisposableEffect(cle) { ...; onDispose { ... } } : du code à
//     l'entrée, et son nettoyage (onDispose) à la sortie.
// Une horloge lance une minuterie quand elle apparaît... et doit

fun heureCourante(): String = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm:ss"))

@Composable
fun Horloge(onJournal: (String) -> Unit, couleur: Color = Color(0xFF3F51B5)) {
    var heure by remember { mutableStateOf(heureCourante()) }
    // la DERNIÈRE fonction onJournal reçue (les effets ci-dessous durent
    // plus longtemps qu'une exécution de la fonction)
    val journal by rememberUpdatedState(onJournal)

    // l'entrée dans la composition, et la sortie
    DisposableEffect(Unit) {
        journal("Horloge : entrée dans la composition - démarrage de la minuterie")
        onDispose { journal("Horloge : sortie de la composition - minuterie arrêtée") }
    }

    // la minuterie : une coroutine, ANNULÉE automatiquement à la sortie
    LaunchedEffect(Unit) {
        while (true) {
            delay(1000) // attendre une seconde, sans bloquer le téléphone
            heure = heureCourante()
        }
    }

    // un paramètre qui change : l'effet est relancé (sauf la première fois)
    var premiereCouleur by remember { mutableStateOf(true) }
    LaunchedEffect(couleur) {
        if (premiereCouleur) premiereCouleur = false else journal("Horloge : recomposée avec une nouvelle couleur")
    }

    Text(heure, fontSize = 40.sp, fontFamily = FontFamily.Monospace, color = couleur)
}

@Composable
fun PageCycleDeVie() {
    var avecHorloge by remember { mutableStateOf(true) }
    var rouge by remember { mutableStateOf(false) }
    val journal = remember { mutableStateListOf<String>() }
    var recherche by remember { mutableStateOf("") }
    var afficherRecherche by remember { mutableStateOf(false) }

    fun tracer(message: String) {
        journal.add("${heureCourante()} $message")
    }

    // la vie de l'activité : on s'ABONNE à ses changements d'état à l'entrée
    // dans la composition, on se DÉSABONNE à la sortie (onDispose)
    val cycleDeVie = LocalLifecycleOwner.current.lifecycle
    DisposableEffect(cycleDeVie) {
        val observateur = LifecycleEventObserver { _, evenement -> tracer("Activité : ${evenement.name}") }
        cycleDeVie.addObserver(observateur)
        onDispose { cycleDeVie.removeObserver(observateur) }
    }

    TitreTache(if (recherche.isEmpty()) "Exemple 13" else "Recherche : $recherche")

    PageExemple("Exemple 13 - le cycle de vie") {
        Card {
            Column(Modifier.padding(12.dp)) {
                LigneCase("afficher l'horloge", avecHorloge) { avecHorloge = it }
                LigneCase("en rouge", rouge) { rouge = it }
                if (avecHorloge) {
                    Horloge(onJournal = { tracer(it) }, couleur = if (rouge) Color.Red else Color(0xFF3F51B5))
                }
            }
        }
        Card {
            Column(Modifier.padding(12.dp)) {
                if (!afficherRecherche) {
                    OutlinedButton(onClick = { afficherRecherche = true }) { Text("Rechercher...") }
                } else {
                    // un champ qui prend le focus dès son apparition
                    val focus = remember { FocusRequester() }
                    OutlinedTextField(
                        value = recherche,
                        onValueChange = { recherche = it },
                        placeholder = { Text("le titre suit la saisie") },
                        modifier = Modifier.fillMaxWidth().focusRequester(focus),
                    )
                    LaunchedEffect(Unit) { focus.requestFocus() }
                }
            }
        }
        Text("Journal", style = MaterialTheme.typography.titleSmall)
        journal.forEachIndexed { i, ligne -> Text("${i + 1}. $ligne", fontSize = 12.sp) }
    }
}

// Comment partager des données et des fonctions entre des composants
// éloignés, sans les passer de paramètre en paramètre à travers tous les
// intermédiaires ? Avec une CompositionLocal, une donnée "locale à la
// composition" :
//   1. on la DÉCLARE une fois : val LocalJournal = staticCompositionLocalOf<Journal> { ... } ;
//   2. un composant placé haut dans l'arbre la FOURNIT à tous ses
//      descendants : CompositionLocalProvider(LocalJournal provides journal) { ... } ;
//   3. n'importe quel descendant, à n'importe quelle profondeur, la LIT :
//      LocalJournal.current.
// C'est le mécanisme que Compose utilise lui-même : MaterialTheme.colorScheme,
// LocalContext.current, LocalDensity.current... sont des lectures de
// CompositionLocal.
//
// Le journal lui-même est un objet ordinaire, dont l'état (ses lignes) est
// OBSERVÉ par Compose (mutableStateListOf) : un composant qui lit ses
// lignes est recomposé quand elles changent.

package fr.stahe.exemples.ex14.fournisseurs

import androidx.compose.runtime.mutableStateListOf
import androidx.compose.runtime.staticCompositionLocalOf
import java.time.LocalTime
import java.time.format.DateTimeFormatter

class Journal {
    // la liste observée ; dehors, on ne la voit que comme une List NON modifiable
    private val _lignes = mutableStateListOf<String>()
    val lignes: List<String> get() = _lignes

    fun tracer(message: String) {
        _lignes.add(0, "${LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm:ss"))} $message")
        if (_lignes.size > 6) _lignes.removeAt(_lignes.lastIndex)
    }
}

@Composable
fun Surligneur(couleur: Color = Color(0xFFFFF3CD), contenu: @Composable () -> Unit) {
    var touche by remember { mutableStateOf(false) }
    // le survol par une souris : une "source d'interactions", dont on
    // observe l'état "survolé"
    val interactions = remember { MutableInteractionSource() }
    val survole by interactions.collectIsHoveredAsState()
    // un changement de couleur ANIMÉ (200 ms)
    val fond by animateColorAsState(if (touche || survole) couleur else Color.Transparent, tween(200), label = "fond")
    Box(
        Modifier
            .background(fond)
            .hoverable(interactions)
            // les contacts du doigt, au plus bas niveau : le doigt se pose,
            // puis se lève (ou le geste est annulé). requireUnconsumed = false :
            // même si le contenu (un bouton) a déjà traité l'appui
            .pointerInput(Unit) {
                awaitEachGesture {
                    awaitFirstDown(requireUnconsumed = false)
                    touche = true
                    waitForUpOrCancellation()
                    touche = false
                }
            },
    ) {
        contenu()
    }
}

import androidx.compose.ui.graphics.Color
import androidx.compose.ui.input.pointer.pointerInput
import androidx.compose.ui.unit.dp

@Composable
fun BoutonAppuiLong(dureeMs: Int = 1000, onAction: () -> Unit, contenu: @Composable () -> Unit) {
    val progression = remember { Animatable(0f) }
    // la portée des coroutines du composant : elles sont annulées s'il disparaît
    val portee = rememberCoroutineScope()
    val action by rememberUpdatedState(onAction)

    Row(
        verticalAlignment = Alignment.CenterVertically,
        modifier = Modifier.pointerInput(Unit) {
            detectTapGestures(
                // onPress : le doigt se pose ; tryAwaitRelease ATTEND qu'il se lève
                onPress = {
                    val remplissage = portee.launch {
                        progression.animateTo(1f, tween(dureeMs, easing = LinearEasing))
                        // l'animation est allée au bout : l'appui a duré assez longtemps
                        progression.snapTo(0f)
                        action()
                    }
                    tryAwaitRelease()
                    // le doigt se lève trop tôt : on arrête le remplissage, l'anneau se vide
                    if (remplissage.isActive) {
                        remplissage.cancel()
                        portee.launch { progression.animateTo(0f, tween(200)) }
                    }
                },
            )
        },
    ) {
        CircularProgressIndicator(
            progress = { progression.value },
            modifier = Modifier.size(22.dp),
            color = Color.Red,
            strokeWidth = 3.dp,
            trackColor = Color(0xFFFFEBEE),
        )
        Spacer(Modifier.width(8.dp))
        contenu()
    }
}

// ex16/Confirmation.kt - « Êtes-vous sûr ? »
// -------------------------------------------------------------------------
// Avec Compose, une fenêtre (AlertDialog) est un composant comme un autre :
// elle est affichée tant qu'un état le demande. Pour poser une question
// et ATTENDRE la réponse, on écrit une petite classe, Confirmation, dont
// la fonction confirmer est une fonction SUSPENDUE (suspend, cf. exemple
// 19) : elle ouvre la fenêtre, puis attend, sans bloquer le téléphone, que
// l'utilisateur réponde. Le code appelant s'écrit alors très simplement,
// dans une coroutine :
//
//     if (confirmation.confirmer("Supprimer ce médecin ?")) { ... }
//
// Fermer la fenêtre SANS répondre (appui en dehors, bouton Retour du
// téléphone) est traité comme un "non".
// La fenêtre elle-même est affichée par FenetreConfirmation(confirmation),
// placée une fois dans l'écran.
// =========================================================================
package fr.stahe.exemples.ex16

import androidx.compose.material3.AlertDialog
import androidx.compose.material3.Button
import androidx.compose.material3.ButtonDefaults
import androidx.compose.material3.Text
import androidx.compose.material3.TextButton
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.compose.ui.graphics.Color
import kotlinx.coroutines.CompletableDeferred

class Confirmation {
    // la question posée (null : pas de fenêtre)
    var question by mutableStateOf<String?>(null)
        private set

    // la réponse attendue : une valeur qui arrivera PLUS TARD
    private var reponse: CompletableDeferred<Boolean>? = null

    suspend fun confirmer(message: String): Boolean {
        val attente = CompletableDeferred<Boolean>()
        reponse = attente
        question = message // la fenêtre s'affiche
        try {
            return attente.await() // la coroutine est suspendue jusqu'à la réponse
        } finally {
            question = null // la fenêtre disparaît
        }
    }

    fun repondre(oui: Boolean) {
        reponse?.complete(oui)
    }
}

// remember : la Confirmation survit aux recompositions de l'écran
@Composable
fun rememberConfirmation(): Confirmation = remember { Confirmation() }

@Composable
fun FenetreConfirmation(confirmation: Confirmation) {
    // ?: return : pas de question, rien à afficher
    val question = confirmation.question ?: return
    // AlertDialog : une fenêtre Material, avec titre, contenu et boutons
    AlertDialog(
        onDismissRequest = { confirmation.repondre(false) }, // fermée sans réponse : "non"
        title = { Text("Confirmation") },
        text = { Text(question) },
        dismissButton = { TextButton(onClick = { confirmation.repondre(false) }) { Text("Non, revenir") } },
        confirmButton = {
            Button(
                onClick = { confirmation.repondre(true) },
                colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFD32F2F)),
            ) { Text("Oui, confirmer") }
        },
    )
}

    • lignes 42-49 : [CompletableDeferred] : une valeur qui arrivera PLUS TARD. [confirmer] affiche la fenêtre, puis ATTEND la réponse ([await()]), sans bloquer le téléphone ; [finally] : dans tous les cas, la fenêtre disparaît ;
    • lignes 53-54 : un bouton de la fenêtre FOURNIT la réponse : la coroutine en attente reprend ;

@Composable
fun PageContacts() {
    val contacts = remember { mutableStateListOf(Contact(1, "Jean MARTIN"), Contact(2, "Alice DURAND"), Contact(3, "Luc GERMAN")) }
    var message by remember { mutableStateOf("") }
    val confirmation = rememberConfirmation()
    // la zone des messages éphémères, et une portée pour lancer des coroutines
    val messages = remember { SnackbarHostState() }
    val portee = rememberCoroutineScope()

    // une fonction suspendue : elle "attend" la réponse de l'utilisateur
    suspend fun supprimer(c: Contact) {
        if (!confirmation.confirmer("Supprimer le contact ${c.nom} ?")) {
            message = "Suppression abandonnée."
            return
        }
        val position = contacts.indexOf(c)
        contacts.remove(c)
        message = "${c.nom} a été supprimé."
        // un message éphémère au bas de l'écran ; showSnackbar attend sa
        // disparition, et dit si son bouton a été touché
        val resultat = messages.showSnackbar("${c.nom} supprimé", actionLabel = "Annuler", duration = SnackbarDuration.Short)
        if (resultat == SnackbarResult.ActionPerformed) {
            contacts.add(position, c)
            message = "${c.nom} est revenu."
        }
    }

    suspend fun toutSupprimer() {
        if (confirmation.confirmer("Supprimer les ${contacts.size} contacts ?")) {
            contacts.clear()
            message = "Tous les contacts ont été supprimés."
        }
    }

    PageExemple("Exemple 16 - une fenêtre de confirmation", messages = messages) {
        for (c in contacts) {
            ListItem(
                leadingContent = { Icon(Icons.Default.Person, contentDescription = null) },
                headlineContent = { Text(c.nom) },
                trailingContent = {
                    // launch : une coroutine, pour pouvoir appeler la fonction suspendue
                    IconButton(onClick = { portee.launch { supprimer(c) } }) {
                        Icon(Icons.Outlined.Delete, contentDescription = "supprimer ${c.nom}", tint = Color.Red)
                    }
                },
            )
        }
        if (contacts.isEmpty()) {
            ListItem(headlineContent = { Text("Aucun contact.", color = Color.Gray) })
        }
        Spacer(Modifier.height(12.dp))
        Button(
            onClick = { portee.launch { toutSupprimer() } },
            // le bouton est désactivé quand la liste est vide
            enabled = contacts.isNotEmpty(),
            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFD32F2F)),
        ) { Text("Tout supprimer") }
        Spacer(Modifier.height(12.dp))
        if (message.isNotEmpty()) {
            Text(message, color = Color(0xFF3F51B5))
        }
    }
    // la fenêtre, quand une question est posée
    FenetreConfirmation(confirmation)
}

    • lignes 64-77 : une fonction suspendue qui se lit comme du code ordinaire : on demande, on attend la réponse ; on supprime ; on affiche le message, on ATTEND sa disparition ([showSnackbar] dit si son bouton a été touché) et l'on annule si besoin ;
