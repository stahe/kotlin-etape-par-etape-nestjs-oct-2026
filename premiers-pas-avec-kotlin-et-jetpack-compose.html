    3.1.1. Le projet

    Le projet des exemples a été créé par [Android Studio] (modèle [Empty Activity], cf. chapitre précédent), puis complété. Un projet [Android] est un projet [Gradle] : l'outil qui télécharge les bibliothèques, compile le code [Kotlin] et fabrique le fichier [.apk] installé sur le téléphone. Il est décrit par quelques fichiers écrits en [Kotlin] (les fichiers [.kts], « Kotlin script »).

    // la version de l'API Android utilisée pour COMPILER

        version = release(37)
    }

    defaultConfig {
        // l'identifiant UNIQUE de l'application sur le téléphone
        applicationId = "fr.stahe.exemples"
        // la plus ancienne version d'Android acceptée : Android 8.0 (API 26)
        minSdk = 26
        // la version d'Android pour laquelle l'application a été testée
        targetSdk = 37
        versionCode = 1
        versionName = "1.0"
        // ajout : l'adresse du serveur JSON (exemples 22 à 25), modifiable au
        // moment de la compilation : gradlew installDebug -Pserveur=http://192.168.1.20:8080

// l'application. Avec Jetpack Compose, l'activité ne fait presque rien :

    // de la fonction... et même quand Android recrée l'activité (rotation

    Scaffold(

                    "Ex01Activity et de la fonction PageAccueil.",

            Spacer(Modifier.height(8.dp))
            // ${expression} insère la valeur d'une expression dans une chaîne
            Text("On y écrit du ${langages.joinToString(", ")} (${langages.size} technologies).")
            Spacer(Modifier.height(16.dp))
            // onClick : la fonction (ici une "lambda") appelée quand on appuie.

            Button(onClick = {
                compteur++
                heure = heureCourante()

                Text("Cliquez-moi")

            Spacer(Modifier.height(16.dp))

@Composable
fun PageCalculateur() {
    // l'état : ce que l'utilisateur a saisi ou choisi
    var libelle by rememberSaveable { mutableStateOf("Stylo") }
    var prixTexte by rememberSaveable { mutableStateOf("2.5") } // le texte du champ, tel que tapé
    var quantite by rememberSaveable { mutableIntStateOf(1) }
    var tauxTVA by rememberSaveable { mutableDoubleStateOf(20.0) }
    // l'historique des 5 derniers changements du total TTC : une liste NON
    // modifiable, remplacée par une nouvelle liste à chaque changement
    var historique by rememberSaveable { mutableStateOf(listOf<String>()) }

    // des valeurs CALCULÉES : de simples variables locales, recalculées à
    // chaque exécution de la fonction
    val prixHT = prix(prixTexte)
    val totalHT = quantite * prixHT
    val montantTVA = totalHT * tauxTVA / 100
    val ttc = totalHT + montantTVA

    // TOUS les changements de l'état passent par cette fonction locale : elle
    // note le total TTC avant le changement, applique le changement, et
    // complète l'historique si le total a changé
    fun changer(changement: () -> Unit) {
        val ancien = ttc
        changement()
        // les variables d'état ont DÉJÀ leur nouvelle valeur
        val nouveau = totalTTC(prix(prixTexte), quantite, tauxTVA)
        if (nouveau != ancien) {
            historique = (listOf("${euros.format(ancien)} -> ${euros.format(nouveau)}") + historique).take(5)
        }
    }

    // le titre de l'application SUIT l'état, comme tout le reste
    TitreTache("TTC : ${euros.format(ttc)}")
    val styleTotal = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold)

    PageExemple("Exemple 02 - l'état") {
        Card {
            Column(Modifier.padding(16.dp)) {
                // un champ CONTRÔLÉ : il affiche value, et appelle onValueChange
                // à chaque frappe, avec le nouveau texte
                OutlinedTextField(
                    value = libelle,
                    // changer d'article remet la quantité à 1 : on le fait là où
                    // l'on sait POURQUOI l'état change
                    onValueChange = { nouveau ->
                        changer {
                            libelle = nouveau
                            quantite = 1
                        }
                    },
                    label = { Text("Article") },
                    modifier = Modifier.fillMaxWidth(),
                )
                OutlinedTextField(
                    value = prixTexte,
                    onValueChange = { texte -> changer { prixTexte = texte } },
                    label = { Text("Prix unitaire HT") },
                    // le clavier numérique du téléphone
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                    modifier = Modifier.fillMaxWidth(),
                )
                Spacer(Modifier.height(12.dp))
                Row(verticalAlignment = Alignment.CenterVertically) {
                    // weight(1f) : le texte prend toute la largeur restante

    var saisie by remember { mutableStateOf("") }
    var prochainId by remember { mutableIntStateOf(3) }
    // le FocusRequester : de quoi donner le focus au champ (le curseur et le
    // clavier du téléphone) depuis le code
    val focus = remember { FocusRequester() }

    fun tracer(message: String) {
        val heure = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm"))
        journal.add(0, "$heure $message")
    }

    // appelée par le bouton ET par la touche "OK" du clavier (keyboardActions)
    fun ajouter() {
        val libelle = saisie.trim()
        if (libelle.isEmpty()) {
            return
        }
        taches.add(Tache(prochainId++, libelle))
        tracer("ajout de « $libelle »")
        saisie = ""
        // le focus revient au champ : on peut taper la tâche suivante
        focus.requestFocus()
    }

    fun basculer(tache: Tache, faite: Boolean) {
        // on REMPLACE la tâche par une copie modifiée
        taches[taches.indexOf(tache)] = tache.copy(faite = faite)
        tracer("« ${tache.libelle} » : ${if (faite) "faite" else "à faire"}")
    }

    fun supprimer(tache: Tache, comment: String) {
        taches.remove(tache)
        tracer("suppression de « ${tache.libelle} » ($comment)")
    }

    // la densité de l'écran : pour convertir des pixels en dp
    val densite = LocalDensity.current.density

    PageExemple("Exemple 04 - les événements") {
        Row(verticalAlignment = Alignment.CenterVertically) {
            OutlinedTextField(
                value = saisie,
                onValueChange = { saisie = it },
                placeholder = { Text("nouvelle tâche (Échap pour vider)") },
                singleLine = true,
                // la touche "OK" du clavier du téléphone ajoute la tâche
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                keyboardActions = KeyboardActions(onDone = { ajouter() }),
                modifier = Modifier
                    .weight(1f)
                    .focusRequester(focus)
                    // un clavier physique (émulateur, tablette) : Échap vide le champ
                    .onPreviewKeyEvent { evenement ->
                        if (evenement.key == Key.Escape) {
                            saisie = ""
                            true // l'événement est traité
                        } else {
                            false // l'événement poursuit son chemin
                        }
                    },
            )
            Spacer(Modifier.width(8.dp))
            Button(onClick = { ajouter() }) { Text("Ajouter") }
        }
        Spacer(Modifier.height(12.dp))
        for (t in taches) {
            // key(t.id) : l'IDENTITÉ de la ligne. L'état de chaque ligne (son
            // glissement) suit SA tâche, même si la liste change (cf. le cours)
            key(t.id) {
                LigneTache(
                    tache = t,
                    onBasculer = { faite -> basculer(t, faite) },
                    onSupprimer = { comment -> supprimer(t, comment) },
                )
            }
        }
        Spacer(Modifier.height(12.dp))
        // pointerInput : un composant quelconque devient sensible aux gestes
        Box(
            contentAlignment = Alignment.Center,
            modifier = Modifier
                .fillMaxWidth()
                .height(80.dp)
                .background(Color(0xFFE7F5FF), RoundedCornerShape(8.dp))
                .pointerInput(Unit) {
                    detectTapGestures(
                        // la position de l'appui, relative au composant (en pixels)
                        onPress = { p -> tracer("appui en (${(p.x / densite).roundToInt()}, ${(p.y / densite).roundToInt()})") },
                        onDoubleTap = { tracer("double appui") },
                        onLongPress = { tracer("appui long") },
                    )
                },
        ) {
            Text("Touchez cette zone (appui, double appui, appui long)")
        }
        Spacer(Modifier.height(16.dp))
        Text("Journal", style = MaterialTheme.typography.titleSmall)
        for (ligne in journal) {
            Text(ligne, fontSize = 12.sp)
        }
    }
}

// le nom de chaque champ du formulaire
enum class Champ { TITRE, PRENOM, NOM, AGE, NEWSLETTER, EMAIL, PAYS, PRESENTATION }

// les données du formulaire : un objet NON MODIFIABLE (tout est val). Pour
// le "modifier", on en fabrique une copie : profil.copy(nom = "DUPONT"). La
// fonction copy est écrite par Kotlin pour toute classe "data"
data class Profil(
    val titre: String,
    val prenom: String,
    val nom: String,
    val age: Int,
    val newsletter: Boolean,
    val email: String,
    val pays: String,
    val presentation: String,
) {
    // la valeur d'un champ, désigné par son nom
    fun valeur(c: Champ): Any = when (c) {
        Champ.TITRE -> titre
        Champ.PRENOM -> prenom
        Champ.NOM -> nom
        Champ.AGE -> age
        Champ.NEWSLETTER -> newsletter
        Champ.EMAIL -> email
        Champ.PAYS -> pays
        Champ.PRESENTATION -> presentation
    }

    // une copie dont UN champ, désigné par son nom, a une nouvelle valeur
    fun avec(c: Champ, v: Any): Profil = when (c) {
        Champ.TITRE -> copy(titre = v as String)
        Champ.PRENOM -> copy(prenom = v as String)
        Champ.NOM -> copy(nom = v as String)
        Champ.AGE -> copy(age = v as Int)
        Champ.NEWSLETTER -> copy(newsletter = v as Boolean)
        Champ.EMAIL -> copy(email = v as String)
        Champ.PAYS -> copy(pays = v as String)
        Champ.PRESENTATION -> copy(presentation = v as String)
    }

    // l'objet JSON du profil : un champ JSON par champ du formulaire
    fun versJson(): JsonObject = JsonObject(
        Champ.entries.associate { c ->
            val v = valeur(c)
            c.name.lowercase() to when (v) {
                is Int -> JsonPrimitive(v) // "is" : un test de type ; v est ensuite un Int
                is Boolean -> JsonPrimitive(v)
                else -> JsonPrimitive(v.toString())
            }
        },
    )
}

// l'état complet : les valeurs, et les champs déjà quittés
data class Etat(val valeurs: Profil, val touches: Set<Champ>)

// les actions possibles. "sealed" : il n'y en a pas d'autres que celles
// déclarées ici, le compilateur le sait
sealed interface ActionProfil {
    data class Modifier(val champ: Champ, val valeur: Any) : ActionProfil
    data class Quitter(val champ: Champ) : ActionProfil
    data object Majuscules : ActionProfil // un objet unique : l'action n'a pas de données
    data object Reinitialiser : ActionProfil
}

// le réducteur : il ne MODIFIE jamais l'état reçu, il en renvoie un nouveau.
// Dans chaque branche du when, action a le type de la branche ("smart cast") :
// action.champ est accessible pour un Modifier
fun reducteur(etat: Etat, action: ActionProfil): Etat = when (action) {
    is ActionProfil.Modifier -> etat.copy(valeurs = etat.valeurs.avec(action.champ, action.valeur))
    is ActionProfil.Quitter -> etat.copy(touches = etat.touches + action.champ)
    ActionProfil.Majuscules -> etat.copy(
        valeurs = etat.valeurs.copy(prenom = etat.valeurs.prenom.uppercase(), nom = etat.valeurs.nom.uppercase()),
    )
    ActionProfil.Reinitialiser -> Etat(depart, emptySet())
}

val titres = listOf("Mr", "Mme", "Melle")
val formatLogin = Regex("^[a-z0-9._-]{1,30}$")
const val MOT_DE_PASSE_MIN = 6

// les règles : une fonction pure, qui ne dépend que de son argument. Elle
// renvoie, pour chaque champ en erreur, son message. buildMap : on remplit
// une Map modifiable, la fonction renvoie une Map NON modifiable
fun valider(c: Compte): Map<Champ, String> = buildMap {
    if (c.titre !in titres) put(Champ.TITRE, "Choisissez un titre dans la liste.")
    when {
        c.prenom.isBlank() -> put(Champ.PRENOM, "Le prénom est obligatoire.")
        c.prenom.length > 20 -> put(Champ.PRENOM, "Le prénom ne doit pas dépasser 20 caractères.")
    }
    when {
        c.nom.isBlank() -> put(Champ.NOM, "Le nom est obligatoire.")
        c.nom.length > 20 -> put(Champ.NOM, "Le nom ne doit pas dépasser 20 caractères.")
    }
    when {
        c.login.isEmpty() -> put(Champ.LOGIN, "Le login est obligatoire.")
        !formatLogin.matches(c.login) -> put(Champ.LOGIN, "Lettres minuscules non accentuées, chiffres, point, tiret, souligné.")
    }
    when {
        c.password.isEmpty() -> put(Champ.PASSWORD, "Le mot de passe est obligatoire.")
        c.password.length < MOT_DE_PASSE_MIN -> put(Champ.PASSWORD, "Au moins $MOT_DE_PASSE_MIN caractères.")
    }
    // une règle qui compare DEUX champs
    if (c.confirmation != c.password) put(Champ.CONFIRMATION, "La confirmation ne correspond pas au mot de passe.")
}

// typealias : un nom plus parlant pour un type de fonction
typealias Regle = (String) -> String?

// des FABRIQUES de règles : chacune renvoie une fonction de validation
fun obligatoire(message: String): Regle = { v -> if (v.isBlank()) message else null }

// une règle qui en COMPLÈTE une autre : "?:" (l'opérateur Elvis) donne le
// résultat de autre, s'il n'est pas null, sinon celui de la suite
fun maximum(max: Int, autre: Regle): Regle = { v -> autre(v) ?: if (v.length > max) "$max caractères au plus." else null }

class ChampFormulaire(valeurInitiale: String = "", private val regle: Regle = { null }) {
    // l'état du champ : observé par Compose, modifiable seulement ICI (private set)
    var valeur by mutableStateOf(valeurInitiale)
        private set
    var erreur by mutableStateOf<String?>(null)
        private set

    // l'utilisateur a-t-il déjà modifié le champ ?
    private var modifie = false

    // une frappe : le champ se valide dès que l'utilisateur l'a modifié
    fun changer(nouvelle: String) {
        valeur = nouvelle
        modifie = true
        erreur = regle(valeur)
    }

    fun valider(): Boolean {
        erreur = regle(valeur)
        return erreur == null
    }

    // revalider, seulement si l'utilisateur a déjà modifié le champ
    fun revalider() {
        if (modifie) valider()
    }
}

// vararg : un nombre quelconque de champs, Formulaire(prenom, nom, login...)
class Formulaire(private vararg val champs: ChampFormulaire) {
    // map d'abord (TOUS les champs sont validés, et affichent leur erreur),
    // all ensuite (tous sont-ils bons ?)
    fun valider(): Boolean = champs.map { it.valider() }.all { it }
}

// "pipes") : dans une fonction @Composable, on appelle des FONCTIONS
// ordinaires, rangées dans un fichier et testables à part :
//     horaire(creneau)               -> "8:00 - 8:20"
//     horaire(creneau, Partie.DEBUT) -> "8:00"
// Kotlin permet aussi d'AJOUTER des fonctions et des propriétés à un type
// existant, par des EXTENSIONS : "marie pelissier".majusculeInitiale ->
// "Marie Pelissier".
// Pour les dates, les nombres, les montants, les classes de Java
// (java.time, java.text) connaissent les conventions de chaque langue
// (cf. Ex09Activity.kt).

package fr.stahe.exemples.ex09

// un créneau horaire, tel que le décrit l'API RdvMedecins
data class Creneau(val hdebut: Int, val mdebut: Int, val hfin: Int, val mfin: Int)

enum class Partie { TOUT, DEBUT, FIN }

// "8:05" plutôt que "8:5" ; private : invisible hors de ce fichier
private fun heure(h: Int, m: Int) = "$h:${m.toString().padStart(2, '0')}"

// un paramètre FACULTATIF, avec sa valeur par défaut : partie = Partie.TOUT
fun horaire(c: Creneau, partie: Partie = Partie.TOUT): String = when (partie) {
    Partie.DEBUT -> heure(c.hdebut, c.mdebut)
    Partie.FIN -> heure(c.hfin, c.mfin)
    Partie.TOUT -> "${heure(c.hdebut, c.mdebut)} - ${heure(c.hfin, c.mfin)}"
}

// des EXTENSIONS du type String : de nouvelles propriétés pour TOUTES les
// chaînes. Dans le corps, this désigne la chaîne
// "marie pelissier" -> "Marie Pelissier" ; \p{L} : une lettre, accentuée ou non
val String.majusculeInitiale: String
    get() = Regex("\\p{L}+").replace(this) { m ->
        m.value.lowercase().replaceFirstChar { it.uppercase() }
    }

// "Marie PELISSIER" -> "MP"
val String.initiales: String
    get() = split(Regex("\\s+")).filter { it.isNotEmpty() }.joinToString("") { it.first().uppercase() }

// dates : des formats "tout prêts" (SHORT, MEDIUM, LONG, FULL), ou un motif
val dateCourte: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.MEDIUM).withLocale(Locale.FRANCE)
val dateComplete: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.FULL).withLocale(Locale.FRANCE)
val heureSeule: DateTimeFormatter = DateTimeFormatter.ofLocalizedTime(FormatStyle.MEDIUM).withLocale(Locale.FRANCE)
val jourAnglais: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.LONG).withLocale(Locale.US)
val motif: DateTimeFormatter = DateTimeFormatter.ofPattern("dd/MM/yyyy HH:mm", Locale.FRANCE)

// nombres et montants
val euros: NumberFormat = NumberFormat.getCurrencyInstance(Locale.FRANCE)
val entier: NumberFormat = NumberFormat.getNumberInstance(Locale.FRANCE)

// apply : configure l'objet qu'on vient de créer, et le renvoie
val pourcent: NumberFormat = NumberFormat.getPercentInstance(Locale.FRANCE).apply { minimumFractionDigits = 1 }

const val JOUR = "2026-10-05" // une date au format de l'API RdvMedecins
const val PRIX = 1234.5
const val TAUX = 0.055

val langages = listOf("Kotlin", "Compose", "Dart", "Swift", "SQL")
val creneaux = listOf(Creneau(8, 0, 8, 20), Creneau(8, 20, 8, 40), Creneau(14, 5, 14, 25))

@Composable
fun PageMiseEnForme() {
    var maintenant by remember { mutableStateOf(LocalDateTime.now()) }

    // (code affiché, valeur) : une ligne du tableau, sous forme de paire
    val lignes = listOf(
        "dateCourte.format(maintenant)" to dateCourte.format(maintenant),
        "dateComplete.format(maintenant)" to dateComplete.format(maintenant),
        "heureSeule.format(maintenant)" to heureSeule.format(maintenant),
        "motif.format(maintenant)" to motif.format(maintenant),
        "dateComplete.format(LocalDate.parse(JOUR))" to dateComplete.format(LocalDate.parse(JOUR)),
        "jourAnglais.format(LocalDate.parse(JOUR))" to jourAnglais.format(LocalDate.parse(JOUR)),
        "maintenant.toString()" to maintenant.toString(),
        "euros.format(PRIX)" to euros.format(PRIX),
        "entier.format(PRIX)" to entier.format(PRIX),
        "\"%.1f\".format(Locale.ROOT, PRIX)" to "%.1f".format(Locale.ROOT, PRIX),
        "pourcent.format(TAUX)" to pourcent.format(TAUX),
        "NOM.uppercase()" to NOM.uppercase(),
        "NOM.majusculeInitiale" to NOM.majusculeInitiale,
        "\"RDV\".lowercase()" to "RDV".lowercase(),
        "langages.take(3).joinToString(\", \")" to langages.take(3).joinToString(", "),
        "JsonObject(...).toString()" to JsonObject(medecin.mapValues { JsonPrimitive(it.value) }).toString(),
        // deux transformations à la suite : le résultat de la première passe à la seconde
        "NOM.majusculeInitiale.initiales" to NOM.majusculeInitiale.initiales,
    )

    PageExemple("Exemple 09 - la mise en forme") {
        // un tableau : la première colonne prend 55 % de la largeur (weight)
        Row(Modifier.padding(4.dp)) {
            Text("Dans le code", Modifier.weight(0.55f), fontWeight = FontWeight.Bold)
            Text("Affiché", Modifier.weight(0.45f), fontWeight = FontWeight.Bold)
        }
        // une paire se "déstructure" : (code, valeur)
        for ((code, valeur) in lignes) {
            HorizontalDivider()
            Row(Modifier.padding(4.dp)) {
                Text(code, Modifier.weight(0.55f), fontFamily = FontFamily.Monospace, fontSize = 11.sp, color = Color(0xFFC7254E))
                Text(valeur, Modifier.weight(0.45f))
            }
        }
        Spacer(Modifier.height(16.dp))
        Text("Parcourir les entrées d'une Map", style = MaterialTheme.typography.titleSmall)
        for ((cle, valeur) in medecin) {
            Text("$cle = $valeur")
        }
        Spacer(Modifier.height(16.dp))
        Text("Une fonction maison : horaire", style = MaterialTheme.typography.titleSmall)
        for (c in creneaux) {
            Text("${horaire(c)} (début : ${horaire(c, Partie.DEBUT)})")
        }
        Spacer(Modifier.height(16.dp))
        OutlinedButton(onClick = { maintenant = LocalDateTime.now() }) { Text("Actualiser l'heure") }
    }
}
